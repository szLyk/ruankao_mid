1️⃣ 0-1 背包（动态规划 · T0 头牌）

转移方程 dp[i][j] = max(dp[i-1][j], dp[i-1][j-w]+v)　★热区：不放/放两分支、j>=w[i-1] 边界　数据：4 件（重 2/3/4/5，值 3/4/5/6），容量 8　→ 输出 10
Java 版（已实测）
public class Pack { public static void main(String[] args){ int[] w = {2,3,4,5}; // w = 每件物品的重量：件1占2、件2占3、件3占4、件4占5 int[] v = {3,4,5,6}; // v = 每件物品的价值：装进去分别值 3、4、5、6 int C = 8, n = 4; // C = 背包总容量；n = 物品件数 // ★灵魂一行：dp[i][j] = "只准从前 i 件里挑、限重 j" 时能拿到的最大价值 // 行 i = 已考虑前几件；列 j = 背包限重。new int 全 0 → 第 0 行(一件不看)天然为 0 int[][] dp = new int[n+1][C+1]; for(int i = 1; i <= n; i++) // 把第 i 件物品拿到桌面上来决策 for(int j = 0; j <= C; j++){ // 对每种限重档位 j 都重新决策一遍 // ①不拿：答案 = 只看前 i-1 件的旧答案，照抄上一行同列 dp[i][j] = dp[i-1][j]; // ②试拿：j >= w[i-1] 才装得下(w数组从0编号,第i件是w[i-1])； // 拿 = 腾出 w[i-1] 空间(查 dp[i-1][j-w[i-1]]) 再加价值 v[i-1]； // 新成绩 > 不拿 → 覆盖。★两分支取 max 就是这一整个 if if(j >= w[i-1] && dp[i-1][j-w[i-1]] + v[i-1] > dp[i][j]) dp[i][j] = dp[i-1][j-w[i-1]] + v[i-1]; } // 右下角 = "4件随便挑、限重8"的最大价值 → 10（方案：件2+件4） System.out.println(dp[n][C]); } }
C 版
#include <stdio.h> int main(void){ int w[4] = {2,3,4,5}, v[4] = {3,4,5,6}; /* w=重量 v=价值 */ int C = 8, n = 4; /* C=背包总容量 n=件数 */ int dp[5][9]; /* ★灵魂一行：dp[i][j] = 只准从前i件挑、限重j的最大价值 */ for(int j = 0; j <= C; j++) dp[0][j] = 0; /* 第0行：一件不看，价值全0 */ for(int i = 1; i <= n; i++) /* 把第i件拿到桌面上来决策 */ for(int j = 0; j <= C; j++){ /* 对每种限重j都决策一遍 */ dp[i][j] = dp[i-1][j]; /* ①不拿：照抄上一行同列的旧答案 */ /* ②试拿：装得下(j>=w[i-1]) 且 拿了更优(腾出w[i-1]查旧表+价值v[i-1]) */ if(j >= w[i-1] && dp[i-1][j-w[i-1]] + v[i-1] > dp[i][j]) dp[i][j] = dp[i-1][j-w[i-1]] + v[i-1]; } printf("%d\n", dp[n][C]); /* 右下角=答案10（方案：件2+件4） */ return 0; }

2️⃣ LCS 最长公共子序列（DP 家族·字符串类，2014下同族）

状态 dp[i][j] = a 前 i 个与 b 前 j 个的 LCS 长度　★转移：末位相同 dp[i-1][j-1]+1，不同 max(上, 左)　边界：第 0 行/列 = 0　复杂度 O(mn)　→ 输出 4
Java 版（已实测）
public class Lcs { public static void main(String[] args){ char[] a = "ABCBDAB".toCharArray(); // 字符串 a：7 个字符 char[] b = "BDCABA".toCharArray(); // 字符串 b：6 个字符 int m = a.length, n = b.length; // ★灵魂一行：dp[i][j] = a 的前 i 个字符 与 b 的前 j 个字符 的最长公共子序列长度 // (子序列=可以跳着挑但顺序不变；默认第0行/第0列全0：跟空串的LCS自然是0) int[][] dp = new int[m+1][n+1]; for(int i = 1; i <= m; i++) // a的第i个字符 = a[i-1]（下标错位） for(int j = 1; j <= n; j++){ // b的第j个字符 = b[j-1] if(a[i-1] == b[j-1]) dp[i][j] = dp[i-1][j-1] + 1; // ①末位相同 → 接上这个字符 else dp[i][j] = Math.max(dp[i-1][j], dp[i][j-1]); // ②末位不同 → 丢一个，取大 } System.out.println(dp[m][n]); // 右下角 = 4（比如 B C B A） } }
C 版
#include <stdio.h> int main(void){ char a[] = "ABCBDAB", b[] = "BDCABA"; int m = 7, n = 6; int dp[8][7]; /* ★灵魂一行：dp[i][j] = a前i个 与 b前j个 的最长公共子序列长度 */ for(int i = 0; i <= m; i++) dp[i][0] = 0; /* 边界：b为空串 → LCS=0 */ for(int j = 0; j <= n; j++) dp[0][j] = 0; /* 边界：a为空串 → LCS=0 */ for(int i = 1; i <= m; i++) /* a的第i个字符是a[i-1]（下标错位） */ for(int j = 1; j <= n; j++){ /* b的第j个字符是b[j-1] */ if(a[i-1] == b[j-1]) dp[i][j] = dp[i-1][j-1] + 1; /* ①末位相同 → 接上这个字符 */ else{ dp[i][j] = dp[i-1][j]; /* ②末位不同：丢a的末位 */ if(dp[i][j-1] > dp[i][j]) /* 或丢b的末位，谁大取谁 */ dp[i][j] = dp[i][j-1]; } } printf("%d\n", dp[m][n]); /* 右下角 = 4（比如 B C B A） */ return 0; }

3️⃣ 数字三角形（DP 家族·路径类，2017上/2019上同族）

状态 f[i][j] = 从 (i,j) 走到底的最大和　★转移：自底向上，f[i][j] = t[i][j] + max(左下, 右下)　答案 = f[0][0]　复杂度 O(n²)　→ 输出 21
Java 版（已实测）
public class Tri { public static void main(String[] args){ int[][] t = { {2}, {3, 4}, {6, 5, 7}, {4, 1, 8, 3} }; // 2 // 3 4 每个数只能走到下一行的"左下或右下" // 6 5 7 // 4 1 8 3 int n = 4; // ★灵魂一行：f[i][j] = 从位置(i,j)一路走到塔底 能累计的最大和 int[][] f = new int[n][n]; for(int j = 0; j < n; j++) f[n-1][j] = t[n-1][j]; // 底边：人已在底，值=自己（递推起点） for(int i = n - 2; i >= 0; i--) // ★热区：自底向上填表（倒着推） for(int j = 0; j <= i; j++) // 第 i 行只有 i+1 个数 // 站在(i,j)：下一脚要么左下 f[i+1][j]，要么右下 f[i+1][j+1]，挑大的踩 f[i][j] = t[i][j] + Math.max(f[i+1][j], f[i+1][j+1]); System.out.println(f[0][0]); // 塔尖 = 21（路径 2→4→7→8） } }
C 版
#include <stdio.h> int main(void){ int t[4][4] = {{2}, {3,4}, {6,5,7}, {4,1,8,3}}; /* 缺省元素自动补0 */ int f[4][4], n = 4; /* ★灵魂一行：f[i][j] = 从位置(i,j)一路走到塔底能累计的最大和 */ for(int j = 0; j < n; j++) f[n-1][j] = t[n-1][j]; /* 底边：人已在底，值=自己（递推起点） */ for(int i = n - 2; i >= 0; i--) /* ★热区：自底向上填表（倒着推） */ for(int j = 0; j <= i; j++){ /* 第i行只有i+1个数 */ int better = f[i+1][j] > f[i+1][j+1] ? f[i+1][j] : f[i+1][j+1]; f[i][j] = t[i][j] + better; /* ★左下/右下挑大的踩（C无max用三目） */ } printf("%d\n", f[0][0]); /* 塔尖 = 21（路径 2→4→7→8） */ return 0; }

4️⃣ LIS 最长递增子序列（DP 家族·序列类）

状态 f[i] = 以 a[i] 结尾的 LIS 长度　★转移：扫描 i 前面每个 j，能接就 f[i]=max(f[j]+1)　答案 = max(f[i)]——注意答案不在最后一个元素，在全程最大　复杂度 O(n²)　→ 输出 4
Java 版（已实测）
public class Lis { public static void main(String[] args){ int[] a = {3, 1, 4, 1, 5, 9, 2, 6}; int n = a.length, best = 0; // best = 全程见过的最大 f[i] // ★灵魂一行：f[i] = "以 a[i] 结尾"(必须选上a[i])的最长递增子序列长度 // 为什么锁结尾？——只有锁定结尾，才能从前面"比它小"的位置接上去 int[] f = new int[n]; for(int i = 0; i < n; i++){ f[i] = 1; // 保底：自己一个也算长度 1 for(int j = 0; j < i; j++) // 扫描 i 左边每个位置：看能接在谁后面 if(a[j] < a[i] && f[j] + 1 > f[i]) // ★a[j]<a[i] 才可接；接上更长才更新 f[i] = f[j] + 1; if(f[i] > best) best = f[i]; // ★答案 = 所有 f[i] 的最大值（不一定在最后一位！） } System.out.println(best); // 4（如 1 4 5 9 或 3 4 5 9） } }
C 版
#include <stdio.h> int main(void){ int a[8] = {3,1,4,1,5,9,2,6}, n = 8, best = 0; /* best = 全程见过的最大f[i] */ /* ★灵魂一行：f[i] = "以a[i]结尾"(必须选上a[i])的最长递增子序列长度 */ int f[8]; for(int i = 0; i < n; i++){ f[i] = 1; /* 保底：自己一个也算长度1 */ for(int j = 0; j < i; j++) /* 扫描i左边每个位置：能接在谁后面 */ if(a[j] < a[i] && f[j] + 1 > f[i]) /* ★a[j]<a[i]才可接；接上更长才更新 */ f[i] = f[j] + 1; if(f[i] > best) best = f[i]; /* ★答案=所有f[i]的最大值（不一定在最后一位！） */ } printf("%d\n", best); /* 4（如 1 4 5 9 或 3 4 5 9） */ return 0; }

5️⃣ 最大子段和（DP 家族·序列类，Kadane 一遍过）

状态 f = 以当前元素结尾的最大子段和　★转移：max(a[i], f+a[i])——前面攒的是正资产就接，负资产就扔　答案 = 过程最大 f　复杂度 O(n)　→ 输出 6
Java 版（已实测）
public class Ms { public static void main(String[] args){ int[] a = {-2, 1, -3, 4, -1, 2, 1, -5, 4}; int n = a.length; // ★灵魂一行：f = "以当前元素 a[i] 结尾"(必须含a[i])的最大子段和 // ans = 全程出现过的最大 f（真正的答案要靠它保存） int f = a[0], ans = a[0]; for(int i = 1; i < n; i++){ // 决策：前面的积累 f 是正资产 → 接上(f+a[i])；是负资产 → 扔掉，从 a[i] 自立门户 f = Math.max(a[i], f + a[i]); // ★热区：一句话转移 if(f > ans) ans = f; // 每步看一眼，最大值可能出现在任何位置 } System.out.println(ans); // 6（子段 4,-1,2,1） } }
C 版
#include <stdio.h> int main(void){ int a[9] = {-2,1,-3,4,-1,2,1,-5,4}, n = 9; /* ★灵魂一行：f = "以当前元素a[i]结尾"(必须含a[i])的最大子段和；ans = 全程最大f */ int f = a[0], ans = a[0]; for(int i = 1; i < n; i++){ /* 决策：前面是正资产就接上(f+a[i])；是负资产就扔掉，从a[i]自立门户 */ f = f + a[i] > a[i] ? f + a[i] : a[i]; /* ★热区：一句话转移（C无max用三目） */ if(f > ans) ans = f; /* 每步看一眼，最大值可能在任何位置 */ } printf("%d\n", ans); /* 6（子段 4,-1,2,1） */ return 0; }

6️⃣ 二分查找（分治与二分）

前提：数组有序　★热区：lo<=hi、lo=mid+1、hi=mid-1 三处，填空最固定　复杂度 O(logn)　→ 输出 下标=3
Java 版（已实测）
public class Bs { public static void main(String[] args){ int[] a = {1,3,5,7,9,11}; // ★前提：数组必须有序！乱序二分没有意义 int key = 7; // 要找的数 int lo = 0, hi = a.length - 1; // lo/hi = 当前搜索范围的左右边界（闭区间） while(lo <= hi){ // ★热区：范围还有东西（哪怕只剩1个）就继续 int mid = (lo + hi) / 2; // 砍一半，看中间那位 if(a[mid] == key){ // 运气好：中间正好是它 System.out.println("下标=" + mid); return; // 找到就收工 }else if(a[mid] < key) lo = mid + 1; // ★中间比目标小 → 目标只可能在右半，扔掉左半 else hi = mid - 1; // ★中间比目标大 → 扔掉右半 } System.out.println("未找到"); // 走到这里 lo>hi：范围空了 → 不存在 } }
C 版
#include <stdio.h> int main(void){ int a[6] = {1,3,5,7,9,11}; /* ★前提：数组必须有序！乱序二分没有意义 */ int key = 7; /* 要找的数 */ int lo = 0, hi = 5; /* lo/hi = 当前搜索范围的左右边界（闭区间） */ while(lo <= hi){ /* ★热区：范围还有东西（哪怕只剩1个）就继续 */ int mid = (lo + hi) / 2; /* 砍一半，看中间那位 */ if(a[mid] == key){ /* 运气好：中间正好是它 */ printf("下标=%d\n", mid); return 0; /* 找到就收工 */ }else if(a[mid] < key) lo = mid + 1; /* ★中间比目标小 → 目标只可能在右半，扔掉左半 */ else hi = mid - 1; /* ★中间比目标大 → 扔掉右半 */ } printf("未找到\n"); /* 走到这里 lo>hi：范围空了 → 不存在 */ return 0; }

7️⃣ 活动安排（贪心 · 出镜率并列最高）

按结束时间升序排好后，从头贪心：下一个的开始 ≥ 上一个的结束就选　★热区：s[i] >= f[last]　复杂度 O(n)（排序 O(nlogn)）　→ 输出 最多活动数=3
Java 版（已实测）
public class Act { public static void main(String[] args){ int[] s = {1, 3, 2, 5, 8}; // 各活动的开始时间 int[] f = {4, 5, 6, 7, 9}; // 各活动的结束时间 —— ★必须已按结束时间升序（贪心前提） // 贪心策略：结束越早的活动给后面留的空间越多 → 结束早的先选 int count = 1, last = 0; // 第一个(结束最早)无条件先选；last = 最后选中活动的下标 for(int i = 1; i < f.length; i++){ // ★热区：第 i 个活动的开始 ≥ 上一个选中活动的结束 → 不冲突，选它 if(s[i] >= f[last]){ count++; last = i; } } System.out.println("最多活动数=" + count); // 3（选了 (1,4)(5,7)(8,9)） } }
C 版

8️⃣ 斐波那契（递归递推 · 送分板块）

同一算法两副面孔：递归 O(2ⁿ)（出口+自调用）vs 递推 O(n)（打表）　★热区：出口 n<=1、递推项 f[i-1]+f[i-2]　→ 输出 55 55
Java 版（已实测）
public class Fib { static long fibRec(int n){ // 版本A：递归 —— 直接翻译定义 f(n)=f(n-1)+f(n-2) if(n <= 1) return n; // ★热区：出口（n=0返0，n=1返1） return fibRec(n-1) + fibRec(n-2); // 自己调自己两次 → 大量重复计算，O(2^n) } static long fibDp(int n){ // 版本B：递推(DP) —— 从小到大打表，每个只算一次 long[] f = new long[n+2]; // f[] 就是"表"：存中间结果避免重复算 f[0] = 0; f[1] = 1; // 初始条件（=递归的出口） for(int i = 2; i <= n; i++) // 从小到大：算 f[i] 时前两项已现成 f[i] = f[i-1] + f[i-2]; // ★热区：递推式与递归式一模一样 return f[n]; // O(n)：快就快在"算过的不再算" } public static void main(String[] args){ System.out.println(fibRec(10) + " " + fibDp(10)); // 两版答案相同：55 55 } }

public class Rev { public static void main(String[] args){ int[] a = {1,2,3,4,5}; // 双指针：i 从左往右，j 从右往左，向中间走 for(int i = 0, j = a.length - 1; i < j; i++, j--){ // ★热区：i<j 就继续 —— 相遇即停（奇数个时中间那个不用动） int t = a[i]; a[i] = a[j]; a[j] = t; // ★热区：交换三行（借中间变量 t） } for(int x : a) System.out.print(x + " "); // 输出 5 4 3 2 1 System.out.println(); } }

🔟 4 皇后（回溯 · T1）

三要素落地：逐行递归尝试 → ok() 约束检查（同列/对角线）→ for 走完自动隐式回溯　★热区：ok() 与 place() 递归行　→ 输出 4皇后解数=2
Java 版（已实测）
public class Nq { static int n = 4, count = 0; static int[] pos = new int[n]; // pos[i] = 第 i 行皇后放在哪一列（每行恰好一个） static boolean ok(int row, int col){ // ★热区：约束检查 —— (row,col)能放吗？ for(int i = 0; i < row; i++) // 逐个检查上面已放好的每一行 // 两种冲突：同列(pos[i]==col)；对角线(|列差|==|行差|) if(pos[i] == col || Math.abs(pos[i] - col) == row - i) return false; return true; // 跟谁都不冲突 → 可放 } static void place(int row){ // ★热区：逐行递归尝试 —— 一次只处理一行 if(row == n){ count++; return; } // 出口：0~n-1 行全放完 → 找到一解 for(int col = 0; col < n; col++) // 本行从第 0 列试到第 n-1 列 if(ok(row, col)){ // 这一列不冲突 → 放上，去下一行 pos[row] = col; place(row + 1); } // for 自然走完 = 所有列试尽 = ★隐式回溯 } public static void main(String[] args){ place(0); // 从第 0 行开始放 System.out.println("4皇后解数=" + count); // 2 } }
C 版
#include <stdio.h> #include <stdlib.h> int n = 4, count = 0, pos[4]; /* pos[i] = 第i行皇后放在哪一列（每行恰好一个） */ int ok(int row, int col){ /* ★热区：约束检查 —— (row,col)能放吗？ */ for(int i = 0; i < row; i++) /* 逐个检查上面已放好的每一行 */ if(pos[i] == col || abs(pos[i] - col) == row - i) return 0; /* 同列/对角线冲突 */ return 1; } void place(int row){ /* ★热区：逐行递归尝试 —— 一次只处理一行 */ if(row == n){ count++; return; } /* 出口：0~n-1行全放完 → 找到一解 */ for(int col = 0; col < n; col++) /* 本行从第0列试到第n-1列 */ if(ok(row, col)){ /* 这列不冲突 → 放上，去下一行 */ pos[row] = col; place(row + 1); } /* for自然走完 = 所有列试尽 = ★隐式回溯 */ } int main(void){ place(0); /* 从第0行开始放 */ printf("4皇后解数=%d\n", count); /* 2 */ return 0; }

📋 策略判定卡（21号§三~§九）

分治子问题相互独立：归并、快排、二分、汉诺塔。分解→求解→合并三步。
贪心每步局部最优、不回头：活动选择（按结束时间排序）、部分背包（可分割）、哈夫曼构造（每次合并权值最小的两棵树，n 个叶子 → 共 2n−1 个结点）。反例：0-1 背包贪心不保证最优。
动态规划子问题重叠+最优子结构：背包/LCS/LIS/编辑距离/最大子段和/数字三角形。自底向上打表。
回溯试探+回退（深度优先+剪枝）：N 皇后、迷宫、子集枚举。
P / NP / NPC（选择题）P=多项式时间可解；NP=多项式时间可验证；NPC=NP 中最难的一类；P=NP 尚未证明。
近似算法不求最优求"够好"，用于 NPC 问题（如旅行商）；概率算法（随机化）结果可能带概率保证。
